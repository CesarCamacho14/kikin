package com.example.listita; // o listito, dependiendo de cómo llamaste al proyecto

import android.content.Context;
import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.BaseAdapter;
import android.widget.ImageView;
import android.widget.TextView;

public class adaptador extends BaseAdapter {
    Context contexto;
    String[] arreglocomenta;
    String[] arreglolista;
    int[] arregloimagenes;
    LayoutInflater inflater;

    //constructor
    public adaptador(Context contexto, String[] arreglocomenta, String[] arreglolista, int[] arregloimagenes) {
        this.contexto = contexto;
        this.arreglocomenta = arreglocomenta;
        this.arreglolista = arreglolista;
        this.arregloimagenes = arregloimagenes;
        this.inflater = LayoutInflater.from(contexto); //se le movio esto
    }

    @Override
    public int getCount() {
        return arreglocomenta.length; //se le movio esto
    }

    @Override
    public Object getItem(int position) {
        return null;
    }

    @Override
    public long getItemId(int position) {
        return 0;
    }

    @Override
    public View getView(int position, View convertView, ViewGroup parent) {
        convertView = inflater.inflate(R.layout.basurita, null); // esta linea

        TextView texto1 = convertView.findViewById(R.id.textito1);
        TextView texto2 = convertView.findViewById(R.id.textito2);
        ImageView imagencita = convertView.findViewById(R.id.imageicon);

        texto1.setText(arreglolista[position]);
        texto2.setText(arreglocomenta[position]);
        imagencita.setImageResource(arregloimagenes[position]);

        return convertView;
    }
}
