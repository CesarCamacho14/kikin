package com.tu_paquete; // Reemplaza con el nombre de tu paquete

import android.os.Bundle;
import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import androidx.annotation.NonNull;
import androidx.annotation.Nullable;
import androidx.fragment.app.Fragment;

public class EjemploFragment extends Fragment {

    // Necesitamos un constructor público vacío por defecto
    public EjemploFragment() {
    }

    // Este método es el encargado de "inflar" (dibujar) el XML del fragmento
    @Nullable
    @Override
    public View onCreateView(@NonNull LayoutInflater inflater, @Nullable ViewGroup container, @Nullable Bundle savedInstanceState) {
        // Inflamos el diseño fragment_ejemplo.xml
        return inflater.inflate(R.layout.fragment_ejemplo, container, false);
    }
}

        android:layout_width="match_parent"
        android:layout_height="match_parent" />

</LinearLayout>

//en activity main.java
package com.tu_paquete; // Reemplaza con el nombre de tu paquete

import android.os.Bundle;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        // Verificamos que savedInstanceState sea null para no cargar el fragmento 
        // múltiples veces si se rota la pantalla del celular.
        if (savedInstanceState == null) {
            
            // 1. Creamos una instancia de nuestro Fragmento
            EjemploFragment miFragmento = new EjemploFragment();

            // 2. Iniciamos la transacción para agregar el fragmento a la pantalla
            getSupportFragmentManager().beginTransaction()
                    .setReorderingAllowed(true)
                    // Reemplazamos el contenedor vacío con nuestro fragmento
                    .replace(R.id.contenedor_fragmento, miFragmento) 
                    .commit(); // Confirmamos los cambios
        }
    }
}
